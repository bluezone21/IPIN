Jalankan SQL berikut di SQL Editor Supabase Anda untuk membuat tabel:

-- Users (kasir) CREATE TABLE users ( id uuid PRIMARY KEY DEFAULT gen_random_uuid(), username text UNIQUE NOT NULL, password text NOT NULL, name text NOT NULL, role text DEFAULT 'kasir', initials text ); -- Products CREATE TABLE products ( id serial PRIMARY KEY, name text NOT NULL, price int NOT NULL, stock int DEFAULT 0, emoji text DEFAULT '📦', created_at timestamptz DEFAULT now() ); -- Transactions CREATE TABLE transactions ( id serial PRIMARY KEY, code text, total int NOT NULL, method text DEFAULT 'tunai', kasir text, items jsonb, created_at timestamptz DEFAULT now() ); -- Seed users INSERT INTO users (username,password,name,role,initials) VALUES ('admin','admin123','Admin','Administrator','AD'), ('kasir','kasir123','Kasir','Kasir Warung','KS'); -- Seed products INSERT INTO products (name,price,stock,emoji) VALUES ('Indomie Goreng',3500,48,'🍜'), ('Teh Botol Sosro',4500,24,'🍵'), ('Aqua 600ml',3000,60,'💧'), ('Sabun Lifebuoy',7000,15,'🧼'), ('Gula 1kg',14000,20,'🍬'), ('Minyak 1L',18000,10,'🛢️'), ('Rokok Sampoerna',25000,30,'🚬'), ('Kopi Sachet',2500,100,'☕');
Catatan: Aktifkan Row Level Security (RLS) sesuai kebutuhan produksi Anda.
